class UploadState(str, Enum):
    PENDING   = "pending"
    UPLOADING = "uploading"
    DONE      = "done"
    FAILED    = "failed"
    SKIPPED   = "skipped"


class RecordingRepo:
    async def create(self, camera_id: str, filename: str, started_at: datetime,
                     reason: str, trigger_event_id: str | None) -> int: ...
    async def finalize(self, rec_id: int, ended_at: datetime, duration_s: float,
                       size_bytes: int, sha256: str, thumbnail_path: str,
                       upload_state: UploadState) -> None: ...
    async def next_pending(self, limit: int = 5) -> list[Recording]: ...
    async def mark_upload(self, rec_id: int, state: UploadState,
                          drive_file_id: str | None = None,
                          error: str | None = None) -> None: ...
    async def expired_local(self, before: datetime) -> list[Recording]: ...
