```python
@field_validator("yolo_model_path")
@classmethod
def _validate_model_path(cls, v: str) -> str:
    allowed = {".pt", ".onnx"}
    p = Path(v)
    if p.suffix.lower() not in allowed:
        raise ValueError(f"Extension de modelo no permitida: {p.suffix}. Permitidas: {allowed}")
    root = Path(__file__).resolve().parent.parent
    resolved = (root / p).resolve() if not p.is_absolute() else p.resolve()
    if not resolved.is_relative_to(root):
        raise ValueError(f"El modelo debe estar dentro del proyecto: {resolved}")
    return v
```
