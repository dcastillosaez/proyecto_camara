    ```javascript
    let _feedActivated = false;
    export function activateCameraFeed() {
      if (_feedActivated) return;
      _feedActivated = true;
      const img = document.getElementById('camera-feed');
      if (img) img.src = '/video_feed';
    }

    const RTSP_STATES = {
      connected:    { dot: 'bg-green-400', text: 'Conectado' },
      reconnecting: { dot: 'bg-amber-500', text: 'Reconectando…' },
      offline:      { dot: 'bg-red-500',   text: 'Sin señal' },
    };

    export async function loadRtspCard() {
      try {
        const [healthRes, configRes] = await Promise.all([
          fetch('/api/v2/cameras/cam1/health'),
          fetch('/api/v2/config'),
        ]);
        if (!healthRes.ok) throw new Error('health');
        const h = await healthRes.json();
        const state = !h.connected ? 'offline' : (h.last_frame_age_s >= 5 ? 'reconnecting' : 'connected');
        const s = RTSP_STATES[state];
        document.getElementById('rtsp-dot').className = `w-2 h-2 rounded-full ${s.dot}`;
        document.getElementById('rtsp-status-text').textContent = s.text;
        document.getElementById('rtsp-last-frame').textContent = `${h.last_frame_age_s.toFixed(1)} s`;
        document.getElementById('rtsp-reconnects').textContent = `${h.reconnects}`;
        const [w, hh] = h.native_resolution || [0, 0];
        document.getElementById('rtsp-resolution').textContent = w && hh ? `${w}×${hh}` : '–';
        if (configRes.ok) {
          const cfg = await configRes.json();
          const camara = cfg.sections.find(s => s.key === 'camara');
          const captura = camara && camara.groups.find(g => g.key === 'captura');
          const urlField = captura && captura.fields.find(f => f.key === 'camera_url');
          document.getElementById('rtsp-url').textContent = urlField ? urlField.value : '–';
        }
      } catch {
        document.getElementById('rtsp-dot').className = 'w-2 h-2 rounded-full bg-red-500';
        document.getElementById('rtsp-status-text').textContent = 'Sin señal';
      }
    }

    export function tickCameraFooter() {
      const el = document.getElementById('camera-updated');
      if (el) el.textContent = `Actualizado a las ${new Date().toLocaleTimeString('es-ES', { hour12: false })}`;
    }

    export function initCamera() {
      loadRtspCard();
      setInterval(loadRtspCard, 5000);   // mismo ritmo que loadObservability, sin timer nuevo real de negocio
      tickCameraFooter();
      setInterval(tickCameraFooter, 1000);
    }
    ```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-32-12 | Information Disclosure | `camera.js::loadRtspCard` | mitigate | La URL RTSP se lee ya enmascarada de `GET /api/v2/config` (`camera_url`, servidor la enmascara) — el cliente nunca reconstruye ni desenmascara, cero lógica de credenciales en el navegador |
