<objective>
Construir el orquestador de la vista Ajustes: `settings.js` (carga el esquema una vez, pinta el
árbol de 8 secciones como `tablist` vertical anidado, gestiona el cambio de sección y el
deep-link `#ajustes/{sección}`) y `settings-section.js` (pinta el panel de una sección completa:

llame — ninguna fila de las ~88 se pinta.

    ```javascript
    import { renderSection } from './settings-section.js';
    import { isDirty, setOnSectionSaved } from './settings-save.js';

    let _schema = null;   // respuesta completa de GET /api/v2/config
    let _activeSection = 'camara';

    export async function initSettings() {
      setOnSectionSaved((sectionKey) => { _refreshTreeDirtyDots(); if (sectionKey === _activeSection) _openSection(sectionKey); });
      await _loadSchema();
      const hashSection = _sectionFromHash();
      _openSection(hashSection || 'camara');
    }

    async function _loadSchema() {
      _renderSkeleton();
      try {
        const res = await fetch('/api/v2/config');
        if (!res.ok) throw new Error(String(res.status));
        _schema = await res.json();
        _renderTree();
      } catch {
        _renderLoadError();
      }
    }
    ```

    ```javascript
    import { renderField } from './settings-field.js';
    import { trackChange, pendingCount, discardChanges, saveSection, restoreSection } from './settings-save.js';

    export function renderSection(section, sectionKey) {
      const panel = document.getElementById('settings-panel');
      if (!panel) return;
      panel.replaceChildren();   // limpiar sin innerHTML=''
      for (const group of section.groups) {
        if (group.external_source) {
          panel.appendChild(_renderExternalGroup(group));
          continue;
        }
        const fieldset = document.createElement('fieldset');
        fieldset.className = 'mb-6';
        const legend = document.createElement('legend');
        legend.className = 'text-sm font-semibold';
        legend.textContent = group.label;   // servidor -> textContent
        fieldset.appendChild(legend);
        for (const field of group.fields) fieldset.appendChild(renderField(field, sectionKey));
        panel.appendChild(fieldset);
      }
      _renderSaveBar(sectionKey, section);
      _renderRestoreButton(sectionKey, section);
    }
    ```

## Trust Boundaries

| Boundary | Description |
|----------|--------------|
| Servidor (`section.label`/`group.label`) -> `settings.js`/`settings-section.js` | Mismo riesgo que `32-05`: todo el árbol y los fieldsets vienen del esquema servido, nunca hardcodeados. |
| `GET /api/zones` / `GET /api/v2/rules` -> subsecciones de solo lectura | Dos endpoints ya existentes, reutilizados tal cual — ninguna lógica de escritura nueva aquí. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-32-18 | Tampering (stored XSS) | `settings.js` pinta `section.label`/`group.label` en el árbol y en las leyendas de `<fieldset>` | mitigate | `textContent` exclusivamente, igual que `32-05`; verificado en el mismo grep de `innerHTML` del checkpoint de `32-08` |
| T-32-19 | Tampering (stored XSS) | `_renderExternalGroup` pinta nombres de zonas/reglas que vienen de `GET /api/zones`/`GET /api/v2/rules` (datos de usuario, no de `Settings`) | mitigate | Mismo criterio anti-XSS: nombre de zona/regla por `textContent`, nunca interpolado en `innerHTML` |
| T-32-20 | Information Disclosure | Un grupo `external_source` mal configurado apuntara a un endpoint que no exista en el esquema (error de programación, no de usuario) | accept | El `try/catch` de `_renderExternalGroup` degrada a "No se pudo cargar." sin romper el resto del panel ni filtrar detalles del error al usuario |
