```python
class IdentityStateMachine:
    """4 estados de identidad por track y sus transiciones (SPEC_v2.md §5.5, FACE-08).

    Reloj inyectado: ningun metodo llama a time.monotonic(). Un solo hilo
    (RecognitionWorker._loop), por eso no hay lock.

    No construye Event: devuelve IdentityTransition y EventEngine traduce.
    """

    MAX_FAILED_REVALIDATIONS = 3   # criterio 5: tres ciclos de revalidacion (D-04)

    def __init__(
        self,
        voter: TemporalVoter | None = None,
        lost_ttl: float = 30.0,
        revalidate_after: float = 120.0,
        low_confidence: float = 0.55,
    ) -> None:
        self._voter = voter if voter is not None else TemporalVoter()
        self._lost_ttl = lost_ttl
        self._revalidate_after = revalidate_after
        self._low_confidence = low_confidence
        self._states: dict[int, _TrackIdentity] = {}
```

**`state_of(track_id) -> IdentityState`**: devuelve `IdentityState.UNKNOWN` si el track
no está en `_states` (un track nunca visto es UNKNOWN por definición).

**`identity_of(track_id) -> tuple[int | None, float]`**: `(person_id, confidence)` del
estado, o `(None, 0.0)`. Solo devuelve `person_id` si el estado es `CONFIRMED` — un
CANDIDATE aún no tiene identidad asignable.

**`on_face_result(track_id, person_id, score, now) -> IdentityTransition | None`**:
1. `st = self._states.setdefault(track_id, _TrackIdentity())`; `st.last_face_at = now`.
2. `self._voter.vote(track_id, person_id, score)`;
   `winner, conf = self._voter.verdict(track_id)`.
3. `prev = st.state`. Ramificar:

   - **UNKNOWN**: si `person_id is not None` → `st.state = CANDIDATE` y devolver
     `IdentityTransition(..., UNKNOWN, CANDIDATE, emits=False)` (no hay tipo de evento
     para CANDIDATE). Si `person_id is None` → devolver `None`.

   - **CANDIDATE**:
     - Si `winner is not None`: intentar herencia con `self._claim_lost(winner, now)`
       (Task 2). Si hereda → `CONFIRMED` con `emits=False` y `st.recognized_emitted = True`.
       Si no hereda → `CONFIRMED` con `emits = not st.recognized_emitted`, y marcar
       `st.recognized_emitted = True`. En ambos casos `st.person_id = winner`,
       `st.confidence = conf`, `st.failed_revalidations = 0`.
     - Si `winner is None` **y** `self._voter.votes_for(track_id) >= self._voter.window`
       **y** `self._voter.matched_votes(track_id) < self._voter.min_votes` →
       `CANDIDATE → UNKNOWN`, `emits = not st.unknown_emitted`, marcar
       `st.unknown_emitted = True`, `self._voter.reset(track_id)`, `st.person_id = None`.
       **Esta doble condición es la que hace pasar el criterio 3**: mientras sigan
       llegando matches (aunque en conflicto) `matched_votes` >= `min_votes` y el track
       permanece en CANDIDATE. Solo cae a UNKNOWN cuando los matches se agotan.
     - En cualquier otro caso, `None`.

   - **CONFIRMED**: se implementa en la Task 2 (revalidación). En esta task basta con:
     si `winner == st.person_id`, refrescar `st.confidence = conf` y devolver `None`.

   - **TEMPORARILY_LOST**: si `person_id == st.person_id` (un solo match coherente basta:
     la identidad ya se estableció con N votos, exigir la ventana entera dejaría un
     hueco de varios segundos) → `TEMPORARILY_LOST → CONFIRMED`, `st.lost_at = None`,
     `st.failed_revalidations = 0`, `emits=False` (misma visita — FACE-09). Si el match
     es de otra persona, no secuestrar el track: devolver `None`.

4. Rellenar siempre `votes=self._voter.votes_for(track_id)` y
   `window=self._voter.window` en la transición devuelta.

**`on_track_lost(track_id, now) -> IdentityTransition | None`**:
- `st` ausente → `None`.
- `CONFIRMED` → `TEMPORARILY_LOST`, `st.lost_at = now`, `self._voter.reset(track_id)`,
  `emits=False`.
- `CANDIDATE` → eliminar la entrada, `self._voter.reset(track_id)` y devolver
  `CANDIDATE → UNKNOWN` con `emits = not st.unknown_emitted` (D-02: un candidato que
  desaparece sin confirmar es un `UNKNOWN_PERSON`).
- `UNKNOWN` → eliminar la entrada, devolver `None`.
- `TEMPORARILY_LOST` → ya perdido, devolver `None`.

**`on_tick(now) -> list[IdentityTransition]`**: en esta task, solo la expiración de
`lost_ttl`. Para cada entrada en `TEMPORARILY_LOST` con
`st.lost_at is not None and now - st.lost_at > self._lost_ttl`: emitir
`TEMPORARILY_LOST → UNKNOWN` con `emits=True` (IDENTITY_LOST), eliminar la entrada y
`self._voter.reset(track_id)`. Iterar sobre `list(self._states)` para poder borrar
dentro del bucle.

```python
    """Un track nuevo reclama la identidad de un track perdido hace poco.

    ByteTrack nunca reutiliza track_ids: al recuperar a una persona le asigna un id
    nuevo. Sin esta busqueda POR person_id, cada reaparicion arrancaria en UNKNOWN y
    emitiria un segundo PERSON_RECOGNIZED (rompe FACE-09 y FACE-10 a la vez).
    """
    for tid, st in list(self._states.items()):
        if (st.state is IdentityState.TEMPORARILY_LOST
                and st.person_id == person_id
                and st.lost_at is not None
                and now - st.lost_at <= self._lost_ttl):
            del self._states[tid]
            self._voter.reset(tid)
            return True
    return False
```

**`needs_recognition(track_id, now) -> bool`** — el gate de FACE-11:
```python
def needs_recognition(self, track_id: int, now: float) -> bool:
    """A quien merece la pena hacerle inferencia facial ahora mismo (FACE-11).

    Sustituye al filtro `person_id is None` de RecognitionWorker._next_candidate, que
    reintentaba indefinidamente sobre los tracks que nunca llegan a identificarse
    (~120 inferencias/min a 2 FPS, para siempre).
    """
    st = self._states.get(track_id)
    if st is None:
        return True                                    # track nuevo
    if st.state is IdentityState.CANDIDATE:
        return True                                    # votacion en curso
    if st.state is IdentityState.TEMPORARILY_LOST:
        return True                                    # intentar recuperar la identidad
    if st.state is IdentityState.UNKNOWN:
        # Ventana aun sin llenar: seguimos reuniendo evidencia.
        if self._voter.votes_for(track_id) < self._voter.window:
            return True
        # Ventana agotada sin ningun match: backoff. AQUI esta el ahorro del criterio 6.
        return (now - st.last_face_at) >= self._revalidate_after
    # CONFIRMED
    if st.confidence < self._low_confidence:            # D-03: confianza del voter
        return True
    return (now - st.last_face_at) >= self._revalidate_after
```

```python
def on_active_tracks(self, active_ids: set[int], now: float) -> list[IdentityTransition]:
    """Detecta tracks caidos comparando con los ids activos del TrackRegistry.

    Los TEMPORARILY_LOST se saltan: existen precisamente para sobrevivir a que el
    track desaparezca (Pitfall 2 del RESEARCH). El voter SI se poda por active_ids;
    los estados de la FSM no.
    """
    out = []
    for tid in list(self._states):
        if tid in active_ids:
            continue
        if self._states[tid].state is IdentityState.TEMPORARILY_LOST:
            continue
        t = self.on_track_lost(tid, now)
        if t is not None:
            out.append(t)
    self._voter.prune(active_ids)
    return out
```
