Ejemplo del enfoque para `test_no_inference_in_coroutines`:

```python
INFERENCE_CALLS = {"detect_sv", "detect", "embed", "process_crop", "identify_or_register"}

def _async_funcs(tree):
    return [n for n in ast.walk(tree) if isinstance(n, ast.AsyncFunctionDef)]

def test_no_inference_in_coroutines():
    offenders = []
    for path in Path("backend").rglob("*.py"):
        tree = ast.parse(path.read_text(encoding="utf-8"))
        for fn in _async_funcs(tree):
            for node in ast.walk(fn):
                if isinstance(node, ast.Call) and isinstance(node.func, ast.Attribute):
                    if node.func.attr in INFERENCE_CALLS:
                        offenders.append(f"{path}:{node.lineno} {fn.name} -> {node.func.attr}")
    assert not offenders, "Inferencia dentro de corrutinas:\n" + "\n".join(offenders)
```

Estos tests son baratos, rapidos y protegen el invariante de concurrencia mejor que cualquier revision manual. Se ejecutan en cada push.
